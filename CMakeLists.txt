  endif()

    if(NOT EXISTS "${WINAMP_SDK_ROOT}/${SDK_HEADER}")
      message(FATAL_ERROR "SDK fehlt: ${SDK_HEADER}. Siehe docs/build.md.")
    endif()
  endforeach()
  file(SHA256 "${WINAMP_SDK_ROOT}/gen_ml/ml.h" ML_HEADER_HASH)
  file(SHA256 "${WINAMP_SDK_ROOT}/Winamp/wa_ipc.h" IPC_HEADER_HASH)
  file(SHA256 "${WINAMP_SDK_ROOT}/Winamp/ipc_pe.h" PLAYLIST_HEADER_HASH)
  file(SHA256 "${WINAMP_SDK_ROOT}/gen_tray/WINAMPCMD.H" COMMAND_HEADER_HASH)
  file(SHA256 "${WINAMP_SKIN_SDK_ROOT}/gen_ml/ml_ipc_0313.h" SKIN_HEADER_HASH)
  if(NOT ML_HEADER_HASH STREQUAL "afc20588bfcf645c59758fef1ef28b80c55ff9ecc2837e7970f74552998140da" OR

     NOT PLAYLIST_HEADER_HASH STREQUAL "af3abc9f930e1d17bce4e08ac35f848eea0345519d184ee4ea844ddd1660405e" OR
     NOT COMMAND_HEADER_HASH STREQUAL "a6c932f30f418a1bb890eb6e6b1f13b538f2f6aad3632eb24d272e6c175697c6" OR
     NOT SKIN_HEADER_HASH STREQUAL "922e6d21b0b8f2e9fe536456c87e51e05bba82b579ec6eaa77264cbdb25c4e42")
    message(FATAL_ERROR "Die Header entsprechen nicht dem geprüften SDK 5.55 und Skin-Schnittstellenheader von 2016.")
  endif()
