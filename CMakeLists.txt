cmake_minimum_required(VERSION 3.24)
project(WinampPlexLibrary VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(FetchContent)
FetchContent_Declare(nlohmann_json
  URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nlohmann_json)

add_library(plex_security STATIC src/common/security.cpp src/playback/playlist.cpp)
target_include_directories(plex_security PUBLIC include)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive- /EHsc)
  target_compile_options(plex_security PRIVATE /W4 /utf-8 /permissive- /EHsc)
endif()

if(WIN32)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Das Plugin benötigt MSVC und die Plattform Win32 (x86).")
  endif()
  enable_language(RC)
  set(WINAMP_SDK_ROOT "${CMAKE_SOURCE_DIR}/.local/sdk" CACHE PATH "Unverändertes Winamp SDK 5.55")
  foreach(SDK_HEADER gen_ml/ml.h gen_ml/ml_ipc_0313.h Winamp/wa_ipc.h gen_tray/WINAMPCMD.H)
    if(NOT EXISTS "${WINAMP_SDK_ROOT}/${SDK_HEADER}")
      message(FATAL_ERROR "SDK fehlt: ${SDK_HEADER}. Siehe docs/build.md.")
    endif()
  endforeach()
  file(SHA256 "${WINAMP_SDK_ROOT}/gen_ml/ml.h" ML_HEADER_HASH)
  file(SHA256 "${WINAMP_SDK_ROOT}/Winamp/wa_ipc.h" IPC_HEADER_HASH)
  file(SHA256 "${WINAMP_SDK_ROOT}/gen_tray/WINAMPCMD.H" COMMAND_HEADER_HASH)
  file(SHA256 "${WINAMP_SDK_ROOT}/gen_ml/ml_ipc_0313.h" SKIN_HEADER_HASH)
  if(NOT ML_HEADER_HASH STREQUAL "afc20588bfcf645c59758fef1ef28b80c55ff9ecc2837e7970f74552998140da" OR
     NOT IPC_HEADER_HASH STREQUAL "38fef5b50b4e028b5f77c0645a0e2a956607d8468c2ecbba3988d700286e60f8" OR
     NOT COMMAND_HEADER_HASH STREQUAL "a6c932f30f418a1bb890eb6e6b1f13b538f2f6aad3632eb24d272e6c175697c6" OR
     NOT SKIN_HEADER_HASH STREQUAL "3a1cb197ba4e32ea856af2e3d83b74c12924371863eb38286829551ed0fbfb96")
    message(FATAL_ERROR "Die SDK-Header entsprechen nicht dem geprüften SDK 5.55.")
  endif()
  add_library(plex_transport STATIC src/playback/proxy.cpp src/plex/http_stream.cpp src/plex/auth.cpp src/config/settings.cpp src/plex/catalog.cpp src/cache/page_cache.cpp)
  target_include_directories(plex_transport PUBLIC include)
  target_compile_definitions(plex_transport PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(plex_transport PUBLIC plex_security nlohmann_json::nlohmann_json winhttp ws2_32 bcrypt crypt32 shell32 ole32)
  add_library(ml_plex SHARED src/plugin/plugin.cpp src/ui/view.cpp src/ui/library_view.cpp src/ui/login.cpp resources/plugin.rc src/plugin/exports.def)
  target_include_directories(ml_plex PRIVATE "${WINAMP_SDK_ROOT}" resources src)
  target_link_libraries(ml_plex PRIVATE plex_transport comctl32)
  set_target_properties(ml_plex PROPERTIES PREFIX "" OUTPUT_NAME "ml_plex")
  if(BUILD_TESTING)
    add_executable(plugin_host_tests tests/plugin_host_tests.cpp)
    target_include_directories(plugin_host_tests PRIVATE "${WINAMP_SDK_ROOT}")
    target_compile_definitions(plugin_host_tests PRIVATE UNICODE _UNICODE)
    add_test(NAME plugin_lifecycle COMMAND plugin_host_tests $<TARGET_FILE:ml_plex>)
    add_executable(proxy_probe tests/proxy_probe.cpp)
    target_link_libraries(proxy_probe PRIVATE plex_transport)
    add_executable(transport_tests tests/transport_tests.cpp)
    target_link_libraries(transport_tests PRIVATE plex_transport)
    add_test(NAME transport COMMAND transport_tests)
    set_tests_properties(transport PROPERTIES TIMEOUT 30)
    add_executable(config_auth_tests tests/config_auth_tests.cpp)
    target_link_libraries(config_auth_tests PRIVATE plex_transport)
    add_test(NAME config_auth COMMAND config_auth_tests)
    add_executable(catalog_tests tests/catalog_tests.cpp)
    target_link_libraries(catalog_tests PRIVATE plex_transport)
    add_test(NAME catalog COMMAND catalog_tests)
    add_executable(catalog_probe tests/catalog_probe.cpp)
    target_link_libraries(catalog_probe PRIVATE plex_transport)
  endif()
endif()

if(BUILD_TESTING)
  add_executable(security_tests tests/security_tests.cpp)
  target_link_libraries(security_tests PRIVATE plex_security)
  add_test(NAME security COMMAND security_tests)
endif()
